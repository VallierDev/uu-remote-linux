cmake_minimum_required(VERSION 3.16)
project(uu_remote_harmony LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(entry SHARED bridge.cc)
target_compile_options(entry PRIVATE -fno-exceptions -Wall -Wextra)
if(NOT UU_RUST_LIBRARY)
  message(FATAL_ERROR "Set UU_RUST_LIBRARY to the HarmonyOS Rust static library")
endif()
target_link_libraries(entry PRIVATE ${UU_RUST_LIBRARY} ace_napi.z native_avscreen_capture native_media_venc native_media_codecbase native_media_core native_window ohinput hilog_ndk.z dl m)
